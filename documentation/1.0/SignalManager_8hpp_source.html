176 template <typename Func, typename... BoundArgs>
177 unsigned int SignalManager::connect(String widgetName, String signalName, Func&& handler, const BoundArgs&... args)
178 {
179 const unsigned int id = generateUniqueId();
180 auto fwdHandler = std::forward<Func>(handler);
181
182 constexpr bool takesCallerArgs = std::is_invocable_v<Func, BoundArgs..., const std::shared_ptr<Widget>&, const String&>;
183 if constexpr (takesCallerArgs)
184 {
185 m_signals[id] = {widgetName,
186 signalName,
187 makeSignalEx([f = fwdHandler, args...](const std::shared_ptr<Widget>& w, const String& s)
188 { std::invoke(f, args..., w, s); })};
189 }
190 else
191 {
192 m_signals[id] = {widgetName, signalName, makeSignal([f = fwdHandler, args...] { std::invoke(f, args...); })};
193 }
194
195 connect(id);
196 return id;
197 }
