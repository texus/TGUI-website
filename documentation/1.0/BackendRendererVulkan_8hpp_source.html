333 void setObjectName(VkObjectType objectType, HandleType handle, const char* name) const
334 {
335 if (!m_functions.vkSetDebugUtilsObjectNameEXT || (handle == VK_NULL_HANDLE))
336 return;
337
338 VkDebugUtilsObjectNameInfoEXT nameInfo = {};
339 nameInfo.sType = VK_STRUCTURE_TYPE_DEBUG_UTILS_OBJECT_NAME_INFO_EXT;
340 nameInfo.objectType = objectType;
341 nameInfo.objectHandle = reinterpret_cast<std::uint64_t>(handle);
342 nameInfo.pObjectName = name;
343 m_functions.vkSetDebugUtilsObjectNameEXT(m_vulkanContext.device, &nameInfo);
344 }
